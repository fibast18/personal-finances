        // Render Filtered Transaction List

        function renderTransactions() {

            const listContainer = document.getElementById('transactionList');

            const emptyState = document.getElementById('emptyState');


            let filtered = transactions.filter(tx => {

                // Type Filter

                if (selectedTypeFilter !== 'all' && tx.type !== selectedTypeFilter) return false;

                // Category Filter

                if (selectedCategoryFilter !== 'all' && tx.category !== selectedCategoryFilter) return false;

                // Search Query

                if (searchQuery) {

                    const matchDesc = (tx.description || '').toLowerCase().includes(searchQuery);

                    const matchCat = tx.category.toLowerCase().includes(searchQuery);

                    const matchAmt = tx.amount.toString().includes(searchQuery);

                    return matchDesc || matchCat || matchAmt;

                }

                return true;

            });


            // Sort by Date Descending

            filtered.sort((a, b) => new Date(b.datetime) - new Date(a.datetime));


            document.getElementById('txCountBadge').innerText = filtered.length;


            if (filtered.length === 0) {

        // Top Category Visual Bar List

        function renderTopCategories() {

            const container = document.getElementById('topCategoryList');

            const expenses = transactions.filter(t => t.type === 'expense');

            const totalExpense = expenses.reduce((sum, t) => sum + t.amount, 0);


            if (totalExpense === 0) {

                container.innerHTML = `<p class="text-xs text-iosGray text-center py-1">No expense data to analyze</p>`;

                return;

            }


            // Group by category

            const categorySums = {};

            expenses.forEach(t => {

                categorySums[t.category] = (categorySums[t.category] || 0) + t.amount;

            });


            // Convert to array and sort top 3

            const sorted = Object.keys(categorySums)

                .map(cat => ({ category: cat, amount: categorySums[cat] }))

                .sort((a, b) => b.amount - a.amount)

                .slice(0, 3);


            container.innerHTML = sorted.map(item => {

                const config = CATEGORY_CONFIG[item.category] || CATEGORY_CONFIG['Other'];

                const pct = ((item.amount / totalExpense) * 100).toFixed(0);


                return `

                    <div class="space-y-1">

                        <div class="flex justify-between text-xs font-medium">

                            <span class="text-iosGray2 flex items-center gap-1.5">

                                <span class="w-2 h-2 rounded-full" style="background-color: ${config.color}"></span>

                                ${item.category}

                            </span>

                            <span class="text-white">${currencySymbol}${item.amount.toLocaleString('en-PH', { minimumFractionDigits: 2, maximumFractionDigits: 2 })} (${pct}%)</span>

                        </div>

                        <div class="w-full bg-neutral-800 h-1.5 rounded-full overflow-hidden">

                            <div class="h-full rounded-full transition-all duration-500" style="width: ${pct}%; background-color: ${config.color}"></div>

                        </div>

                    </div>

                `;

            }).join('');

        }

        function updateChart() {

            if (!chartInstance) return;


            const emptyState = document.getElementById('chartEmptyState');

            const canvas = document.getElementById('spendingChart');


            if (chartViewMode === 'category') {

                const expenses = transactions.filter(t => t.type === 'expense');

                if (expenses.length === 0) {

                    canvas.classList.add('hidden');

                    emptyState.classList.remove('hidden');

                    return;

                }

                canvas.classList.remove('hidden');

                emptyState.classList.add('hidden');


                const categorySums = {};

                expenses.forEach(t => {

                    categorySums[t.category] = (categorySums[t.category] || 0) + t.amount;

                });


                const labels = Object.keys(categorySums);

                const data = labels.map(cat => categorySums[cat]);

                const colors = labels.map(cat => (CATEGORY_CONFIG[cat] ? CATEGORY_CONFIG[cat].color : '#8e8e93'));


                chartInstance.data.labels = labels;

                chartInstance.data.datasets[0].data = data;

                chartInstance.data.datasets[0].backgroundColor = colors;

            } else {

                // Income vs Expense Ratio Chart

                const totalIncome = transactions.filter(t => t.type === 'income').reduce((s, t) => s + t.amount, 0);

                const totalExpense = transactions.filter(t => t.type === 'expense').reduce((s, t) => s + t.amount, 0);


                if (totalIncome === 0 && totalExpense === 0) {

                    canvas.classList.add('hidden');

                    emptyState.classList.remove('hidden');

                    return;

                }

                canvas.classList.remove('hidden');

                emptyState.classList.add('hidden');


                chartInstance.data.labels = ['Income', 'Expenses'];

                chartInstance.data.datasets[0].data = [totalIncome, totalExpense];

                chartInstance.data.datasets[0].backgroundColor = ['#30d158', '#ff453a'];

            }


            chartInstance.update();

        }
